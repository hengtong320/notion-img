  function flashCombo(){
    clearTimeout(state.comboTimer);
    comboText.textContent=`连解 ×${state.combo}`;
    comboGroup.setAttribute("opacity","1");
    comboGroup.animate([{transform:"scale(.82)",opacity:0},{transform:"scale(1.05)",opacity:1},{transform:"scale(1)",opacity:1}],{duration:260,easing:"ease-out"});
    state.comboTimer=setTimeout(()=>comboGroup.setAttribute("opacity","0"),780);
  }

    state.busy=true;
    const [dx,dy]=DIRS[piece.dir];

    sound(tool?"hint":"ok"); vibrate(tool?18:28);
    piece.el.classList.add("removing");
    const distance=260;
    const anim=piece.el.animate([
      {transform:"translate(0px,0px)",opacity:1},
      {transform:`translate(${dx*distance}px,${dy*distance}px)`,opacity:0}
    ],{duration:360,easing:"cubic-bezier(.18,.78,.18,1)",fill:"forwards"});
    try{await anim.finished}catch(_){}
    piece.active=false;
    piece.el.style.display="none";
    state.history.push({zone:zone.index,piece:piece.id});
    state.moves++;
    const now=performance.now();
    state.combo = now-state.lastCorrectAt<1350 ? Math.min(state.combo+1,9) : 1;
    state.lastCorrectAt=now;
    if(state.combo>=2) flashCombo();
