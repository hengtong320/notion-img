  function complete(){
    if(!state.centerReady||state.busy) return;
    state.busy=true; sound("win"); vibrate([30,30,60,40,90]);
    centerGroup.animate([{transform:"scale(1)"},{transform:"scale(1.12)"},{transform:"scale(1)"}],{duration:900,easing:"ease-in-out"});
    const sec=Math.max(1,Math.round((performance.now()-state.startedAt)/1000));
    setTimeout(()=>showPanel({
      seal:"归",title:"四象归真",text:"青龙、白虎、朱雀、玄武四印合一。",
      stats:[["用时",`${Math.floor(sec/60)}:${String(sec%60).padStart(2,"0")}`],["步数",state.moves],["误判",state.mistakes],["评价",state.mistakes===0?"完美":"完成"]],
      actions:[{label:"再解一次",primary:true,onClick:()=>{hidePanel();resetGame();}}]
    }),620);
  }
