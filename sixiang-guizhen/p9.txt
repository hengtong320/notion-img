  function sound(kind="ok"){
    if(!state.sound) return;
    try{
      state.audio ||= new (window.AudioContext||window.webkitAudioContext)();
      const ctx=state.audio, now=ctx.currentTime;
      if(ctx.state==="suspended") ctx.resume();
      const master=ctx.createGain(); master.connect(ctx.destination);
      master.gain.setValueAtTime(.0001,now);
      const dur=kind==="win"?.95:kind==="bad"?.22:kind==="unlock"?.55:.28;
      master.gain.exponentialRampToValueAtTime(kind==="bad"?.055:.105,now+.012);
      master.gain.exponentialRampToValueAtTime(.0001,now+dur);
      const notes = kind==="bad" ? [116] :
                    kind==="unlock" ? [392,523,659] :
                    kind==="win" ? [330,392,523,659] :
                    kind==="hint" ? [523,659] : [330,440];
      notes.forEach((freq,i)=>{
        const o=ctx.createOscillator();
        o.type=kind==="bad"?"square":"sine";
        o.frequency.setValueAtTime(freq,now+i*.075);
        const g=ctx.createGain(); g.gain.value=1/(i+1);
        o.connect(g); g.connect(master);
        o.start(now+i*.075); o.stop(now+dur+i*.075);
      });
    }catch(_){}
  }

  function zoneGeom(zone){
    const {x,y,w,h}=zone.layout;
    const cell=Math.min(w/zone.gridW,h/zone.gridH);
    const ox=x+(w-cell*zone.gridW)/2;
    const oy=y+(h-cell*zone.gridH)/2;
    return {cell,ox,oy};
  }
  function pxy(zone, p){
    const g=zoneGeom(zone);
    return [g.ox+(p[0]+.5)*g.cell,g.oy+(p[1]+.5)*g.cell];
  }

  }
