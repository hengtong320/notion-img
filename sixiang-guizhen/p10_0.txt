  function renderPieces(){
    pieceLayer.innerHTML=""; ghostLayer.innerHTML=""; particleLayer.innerHTML="";
    state.zones.forEach(z=>{
      z.pieces.forEach(p=>{
        const d=pathD(z,p.points);
        e("path",{d,class:"ghost-path"},ghostLayer);
        const g=e("g",{class:"piece",tabindex:"0",role:"button","aria-label":`${z.name}纹线${p.id+1}`},pieceLayer);
        e("path",{d,class:"piece-shadow"},g);
        e("path",{d,class:"piece-line"},g);
        e("path",{d,class:"piece-hit"},g);
        bindPress(g,()=>handlePiece(z,p));
        p.el=g;
      });
    });
    updateAssist();
  }

  function activeCellMap(zone){
    const map=new Map();
    zone.pieces.forEach(p=>{
      if(!p.active) return;
      p.cells.forEach(c=>map.set(cellKey(c),p));
    });
    return map;
  }
