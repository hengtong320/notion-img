}))};
        const zr={...fake};
        const map=new Map();
        zr.pieces.forEach(q=>q.cells.forEach(c=>map.set(cellKey(c),q)));
        const [dx,dy]=DIRS[p.dir];
        const own=new Set(p.cells.map(cellKey));
        let [x,y]=p.cells[p.cells.length-1]; x+=dx;y+=dy;
        let movable=true;
        while(x>=0&&x<z.gridW&&y>=0&&y<z.gridH){
          const k=`${x},${y}`;
          if(!own.has(k)&&(OBST.has(k)||map.has(k))){movable=false;break;}
          x+=dx;y+=dy;
        }
        if(!movable){ok=false;break;}
        active.delete(pid);
      }
    });
    console.info(ok?"关卡验证通过：96条纹线完整可解":"关卡验证失败");
  }
